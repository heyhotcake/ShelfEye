#!/usr/bin/env python3
"""
Unified LED Manager Service
Runs as a single daemon that accepts commands via named pipe
This prevents DMA channel conflicts and ensures proper cleanup
"""

import os
import sys
import json
import time
import signal
import threading
import fcntl
import atexit
from pathlib import Path

try:
    from rpi_ws281x import PixelStrip, Color
    WS2812_AVAILABLE = True
except ImportError:
    WS2812_AVAILABLE = False
    print("WARNING: rpi_ws281x not available", file=sys.stderr)

# Configuration
PIPE_PATH = "/tmp/led_control_pipe"
STATE_FILE = "/tmp/led_state.json"
LOCK_FILE = "/tmp/led_manager.lock"
LOG_FILE = "/tmp/led_manager.log"

class LEDManagerService:
    def __init__(self, pin=18, num_leds=27):
        self.pin = pin
        self.num_leds = num_leds
        self.strip = None
        self.current_state = "off"
        self.current_priority = 0
        self.flash_thread = None
        self.flash_active = False
        self.lock = threading.Lock()
        self.running = True
        
        # Initialize LED strip
        if WS2812_AVAILABLE:
            try:
                self.strip = PixelStrip(
                    num_leds,
                    pin,
                    800000,  # 800kHz
                    10,      # DMA channel
                    False,   # Invert
                    255,     # Brightness
                    0,       # Channel
                    0x00081000  # Strip type
                )
                self.strip.begin()
                self.log("LED strip initialized successfully")
            except Exception as e:
                self.log(f"Failed to initialize LED strip: {e}")
                sys.exit(1)
        
        # Register cleanup
        atexit.register(self.cleanup)
        signal.signal(signal.SIGTERM, self._signal_handler)
        signal.signal(signal.SIGINT, self._signal_handler)
    
    def log(self, message):
        """Log messages to file"""
        timestamp = time.strftime("%Y-%m-%d %H:%M:%S")
        log_msg = f"[{timestamp}] {message}\n"
        with open(LOG_FILE, "a") as f:
            f.write(log_msg)
        print(log_msg.strip(), file=sys.stderr)
    
    def _signal_handler(self, sig, frame):
        """Handle shutdown signals"""
        self.log("Received shutdown signal")
        self.running = False
        self.cleanup()
        sys.exit(0)
    
    def cleanup(self):
        """Clean shutdown"""
        self.log("Cleaning up...")
        with self.lock:
            self.flash_active = False
            if self.flash_thread:
                self.flash_thread.join(timeout=1)
            if self.strip:
                self._set_all_color(0, 0, 0)
                # Note: Don't delete strip object here as it can cause segfault
        self.save_state({"state": "off", "priority": 0})
    
    def _set_all_color(self, r, g, b):
        """Set all LEDs to a color"""
        if not self.strip:
            return False
        try:
            for i in range(self.num_leds):
                self.strip.setPixelColor(i, Color(r, g, b))
            self.strip.show()
            return True
        except Exception as e:
            self.log(f"Error setting color: {e}")
            return False
    
    def save_state(self, state):
        """Save current state to file"""
        try:
            with open(STATE_FILE, "w") as f:
                json.dump(state, f)
        except Exception as e:
            self.log(f"Error saving state: {e}")
    
    def handle_command(self, command):
        """Process LED control commands"""
        try:
            cmd = json.loads(command)
            action = cmd.get("action")
            priority = cmd.get("priority", 0)
            
            self.log(f"Received command: {action} (priority: {priority})")
            
            with self.lock:
                # Check priority
                if priority < self.current_priority and self.current_state == "red_flash":
                    self.log(f"Command denied - lower priority ({priority} < {self.current_priority})")
                    return {"success": False, "reason": "lower_priority"}
                
                # Stop any existing flash
                if self.flash_active and action != "red_flash":
                    self.flash_active = False
                    if self.flash_thread:
                        self.flash_thread.join(timeout=1)
                
                # Execute action
                if action == "white":
                    success = self._set_all_color(255, 255, 255)
                    if success:
                        self.current_state = "white"
                        self.current_priority = priority
                
                elif action == "off":
                    success = self._set_all_color(0, 0, 0)
                    if success:
                        self.current_state = "off"
                        self.current_priority = 0
                
                elif action == "red_flash":
                    pattern = cmd.get("pattern", "slow")
                    success = self._start_flash(pattern)
                    if success:
                        self.current_state = "red_flash"
                        self.current_priority = 100  # Highest priority
                
                elif action == "stop_flash":
                    self.flash_active = False
                    if self.flash_thread:
                        self.flash_thread.join(timeout=1)
                    success = self._set_all_color(0, 0, 0)
                    if success:
                        self.current_state = "off"
                        self.current_priority = 0
                
                elif action == "status":
                    return {
                        "success": True,
                        "state": self.current_state,
                        "priority": self.current_priority,
                        "flash_active": self.flash_active
                    }
                
                else:
                    return {"success": False, "reason": "unknown_action"}
                
                # Save state
                self.save_state({
                    "state": self.current_state,
                    "priority": self.current_priority
                })
                
                return {"success": success, "state": self.current_state}
                
        except Exception as e:
            self.log(f"Error handling command: {e}")
            return {"success": False, "error": str(e)}
    
    def _start_flash(self, pattern):
        """Start flashing thread"""
        self.flash_active = True
        
        if pattern == "fast":
            on_time, off_time = 0.25, 0.25
        elif pattern == "slow":
            on_time, off_time = 1.0, 1.0
        else:
            on_time, off_time = 0.1, 0.9
        
        def flash_loop():
            while self.flash_active:
                self._set_all_color(255, 0, 0)  # Red
                time.sleep(on_time)
                if not self.flash_active:
                    break
                self._set_all_color(0, 0, 0)  # Off
                time.sleep(off_time)
        
        self.flash_thread = threading.Thread(target=flash_loop)
        self.flash_thread.start()
        return True
    
    def run(self):
        """Main service loop"""
        # Create named pipe
        if os.path.exists(PIPE_PATH):
            os.unlink(PIPE_PATH)
        os.mkfifo(PIPE_PATH)
        os.chmod(PIPE_PATH, 0o666)
        
        self.log("LED Manager Service started")
        
        while self.running:
            try:
                # Open pipe for reading (blocks until writer connects)
                with open(PIPE_PATH, "r") as pipe:
                    command = pipe.read().strip()
                    if command:
                        response = self.handle_command(command)
                        # Write response to response file
                        response_file = f"/tmp/led_response_{os.getpid()}.json"
                        with open(response_file, "w") as f:
                            json.dump(response, f)
            except Exception as e:
                self.log(f"Error in main loop: {e}")
                time.sleep(0.1)

def acquire_lock():
    """Ensure only one instance runs"""
    lock_file = open(LOCK_FILE, "w")
    try:
        fcntl.flock(lock_file, fcntl.LOCK_EX | fcntl.LOCK_NB)
        lock_file.write(str(os.getpid()))
        lock_file.flush()
        return lock_file
    except IOError:
        print("Another LED manager is already running", file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    # Acquire exclusive lock
    lock_handle = acquire_lock()
    
    # Parse arguments
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--pin", type=int, default=18)
    parser.add_argument("--num-leds", type=int, default=27)
    args = parser.parse_args()
    
    # Run service
    service = LEDManagerService(args.pin, args.num_leds)
    try:
        service.run()
    finally:
        lock_handle.close()
        if os.path.exists(LOCK_FILE):
            os.unlink(LOCK_FILE)
