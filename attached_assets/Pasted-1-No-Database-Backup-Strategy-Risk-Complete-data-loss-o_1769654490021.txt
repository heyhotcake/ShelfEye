9. Concurrent Camera Access Race Conditions
Risk: Preview starts during calibration → corrupted homography matrix
Impact: All subsequent captures use bad calibration
Issue: 10-second wait in acquireExclusiveLock is heuristic, not guaranteed
Fix:

// server/camera-session-manager.ts
async acquireExclusiveLock(cameraId: string, reason: ExclusiveReason): Promise<void> {
  // Set lock FIRST
  this.locks.set(cameraId, { cameraId, type: 'exclusive', timestamp: Date.now(), reason });
  
  // Poll subprocess manager for active camera processes
  let retries = 0;
  while (retries < 30) { // 30 seconds max
    const activeProcesses = await subprocessManager.getActiveProcesses();
    const cameraProcesses = activeProcesses.filter(p => 
      p.type === 'python' && p.script.includes('camera') && p.purpose.includes(cameraId)
    );
    
    if (cameraProcesses.length === 0) {
      console.log(`[CameraSessionManager] Camera ${cameraId} confirmed idle after ${retries} checks`);
      return; // Safe to proceed
    }
    
    await new Promise(resolve => setTimeout(resolve, 1000));
    retries++;
  }
  
  throw new Error(`Camera ${cameraId} still in use after 30s - aborting exclusive lock`);
}
10. No Idempotency for Webhook/Retry Operations
Risk: Alert queue retry sends duplicate emails
Impact: Alert fatigue, ignored warnings
Issue: AlertRetryQueue retries alerts but doesn't deduplicate
Fix:

// server/services/alert-queue.ts
interface QueuedAlert {
  id: string; // Use deterministic ID: hash(type + timestamp + message)
  idempotencyKey: string; // Store in DB
  // ...
}

async queueEmailAlert(...) {
  const idempotencyKey = crypto.createHash('sha256')
    .update(`${type}-${timestamp}-${message}`)
    .digest('hex');
  
  // Check if already sent
  const existing = await storage.getAlertByIdempotencyKey(idempotencyKey);
  if (existing && existing.status === 'sent') {
    console.log('[AlertQueue] Alert already sent (idempotent), skipping');
    return;
  }
  
  // Queue with idempotency key
  this.queue.set(id, { id, idempotencyKey, ... });
}
